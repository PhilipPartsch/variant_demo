1/* Traction power limit as a torque limit. */
 2#include "autoconf.h"
 3#include "vcu/vcu.h"
 4
 5#define VCU_MAX_TORQUE_NM 3000
 6
[docs] 7// @need: Apply traction power limit, IMPL_VCU_POWER_LIMIT_APPLY, [SWREQ_VCU_POWER_LIMIT_APPLY]
 8int vcu_torque_limit_nm(int vehicle_limit_kw, int bms_limit_kw, int motor_rpm)
 9{
10    int limit_kw = vehicle_limit_kw;
11#if CONFIG_POWERTRAIN__TYPE__BEV
12    if (bms_limit_kw < limit_kw) {
13        limit_kw = bms_limit_kw;
14    }
15#else
16    (void)bms_limit_kw;
17#endif
18    if (motor_rpm <= 0) {
19        return VCU_MAX_TORQUE_NM;
20    }
21    int torque = limit_kw * 9549 / motor_rpm; /* P[kW] * 9549 / n[rpm] = M[Nm] */
22    return torque < VCU_MAX_TORQUE_NM ? torque : VCU_MAX_TORQUE_NM;
23}
